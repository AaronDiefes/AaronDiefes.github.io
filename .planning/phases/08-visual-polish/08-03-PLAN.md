<task type="auto">
  <name>Task 2: Create nav.js with accessible hamburger menu behavior</name>
  <files>docs/assets/js/nav.js</files>
  <action>
Create `docs/assets/js/nav.js` with accessible hamburger menu behavior:

```javascript
(function() {
  'use strict';

  const toggle = document.querySelector('.menu-toggle');
  const menu = document.querySelector('.nav-links');

  if (!toggle || !menu) return;

  // Toggle menu
  toggle.addEventListener('click', function() {
    const isExpanded = this.getAttribute('aria-expanded') === 'true';
    this.setAttribute('aria-expanded', String(!isExpanded));
    menu.hidden = isExpanded;
  });

  // Close on ESC
  document.addEventListener('keydown', function(e) {
    if (e.key === 'Escape' && !menu.hidden) {
      toggle.setAttribute('aria-expanded', 'false');
      menu.hidden = true;
      toggle.focus();
    }
  });

  // Close when clicking outside
  document.addEventListener('click', function(e) {
    if (!toggle.contains(e.target) && !menu.contains(e.target) && !menu.hidden) {
      toggle.setAttribute('aria-expanded', 'false');
      menu.hidden = true;
    }
  });

  // Close when clicking a nav link (mobile)
  menu.querySelectorAll('a').forEach(function(link) {
    link.addEventListener('click', function() {
      if (window.innerWidth <= 768) {
        toggle.setAttribute('aria-expanded', 'false');
        menu.hidden = true;
      }
    });
  });
})();
```

Keep nav.js small and dependency-free. Use an IIFE to avoid global scope pollution.
