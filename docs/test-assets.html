Source Over Blend Mode

C++ implementation of the "source over" blend mode algorithm.

// Blend mode: Source Over
// S + (1 - Sa)*D
GPixel src_over_mode(GPixel src, GPixel dest) {
    int sa = GPixel_GetA(src);
    int sr = GPixel_GetR(src);
    int sg = GPixel_GetG(src);
    int sb = GPixel_GetB(src);

    int da = GPixel_GetA(dest);
    int dr = GPixel_GetR(dest);
    int dg = GPixel_GetG(dest);
    int db = GPixel_GetB(dest);

    // Compute blended result
    int ba = sa + div255((255-sa)*da);
    int br = sr + div255((255-sa)*dr);
    int bg = sg + div255((255-sa)*dg);
    int bb = sb + div255((255-sa)*db);

    return GPixel_PackARGB(ba, br, bg, bb);
}

Linear Gradient Shader

Creating a linear gradient shader with color stops.

// Create linear gradient shader
std::vector<GColor> colors = {
    {1.0f, 0.0f, 0.0f, 1.0f},  // Red
    {0.0f, 0.0f, 1.0f, 1.0f}   // Blue
};
std::vector<float> positions = {0.0f, 1.0f};

GPoint p0 = {0, 0};
GPoint p1 = {width, 0};

auto shader = GCreateLinearGradient(p0, p1, colors, positions);
if (shader) {
    paint.setShader(shader);
    canvas->drawRect(rect, paint);
}

Transformation Matrix

Applying rotation and translation transformations to the canvas.

// Apply transformations
GMatrix matrix;

// Translate to center
matrix = GMatrix::Translate(centerX, centerY);

// Rotate by angle (in radians)
matrix = matrix * GMatrix::Rotate(angle);

// Scale uniformly
matrix = matrix * GMatrix::Scale(scale, scale);

// Apply to canvas
canvas->concat(matrix);

// Draw shape with transformation
canvas->drawRect(rect, paint);
